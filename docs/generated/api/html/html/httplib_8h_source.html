173  Request parseRequest(const std::string& raw) {
174  Request req;
175  std::istringstream stream(raw);
176  std::string line;
177  if (!std::getline(stream, line)) {
178  return req;
179  }
180  std::istringstream request_line(line);
181  request_line >> req.method >> req.path;
182 
183  while (std::getline(stream, line) && !line.empty() && line != "\r") {
184  auto colon = line.find(':');
185  if (colon != std::string::npos) {
186  std::string key = line.substr(0, colon);
187  std::string value = line.substr(colon + 1);
188  while (!value.empty() && (value.front() == ' ' || value.front() == '\t')) {
189  value.erase(value.begin());
190  }
191  if (!value.empty() && value.back() == '\r') {
192  value.pop_back();
193  }
194  req.headers[key] = value;
195  }
196  }
197 
198  if (req.headers.count("Content-Length") > 0) {
199  size_t content_length = std::stoul(req.headers.at("Content-Length"));
200  req.body.resize(content_length);
201  stream.read(&req.body[0], content_length);
202  }
203  return req;
204  }
